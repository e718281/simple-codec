@(TupleSize: Int)

package net.scalax.simple.adt

import instance._

@AppendFunc2(i2: Int, sum: Int) = {
    @if(i2 < sum) {
        Successor[AdtFunction[Target, T@i2], @AppendFunc2(i2+1, sum)]
    } else {
        One[AdtFunction[Target, T@i2]]
    }
}

@for(i1 <- 1 to TupleSize) {

    trait CoProduct@{i1}Apply[@for(i2 <- 1 to i1) { T@i2, }] extends CoProduct@{i1}ApplyHelper[@for(i2 <- 1 to i1) { T@i2, }] { Self =>
        def instance[Target](target: Target)(implicit adtCv: ADTValue[@AppendFunc2(1, i1)]): CoProduct@{i1}[@for(i2 <- 1 to i1) { T@i2, }] =
            typeOnly[Target].fold(@for(i2 <- 1 to i1) { (t => Self.instance@{i2}(t.instance(target))), })

        def typeOnly[Target](implicit adtCv: ADTValue[@AppendFunc2(1, i1)]): CoProduct@{i1}[@for(i2 <- 1 to i1) { AdtFunction[Target, T@i2], }] =
            new CoProduct@{i1}(adtCv.value)
    }

}
